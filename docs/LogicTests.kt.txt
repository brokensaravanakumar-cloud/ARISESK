import com.jinwoo.ai.core.*

var fails = 0
fun check(name: String, got: Any?, want: Any?) {
    if (got != want) { fails++; println("FAIL $name\n   got : $got\n   want: $want") } else println("ok   $name")
}
fun main() {
    check("alarm 6:30 am", CommandParser.parse("Set an alarm for 6:30 AM"), Cmd.SetAlarm(6, 30))
    check("alarm 7 pm", CommandParser.parse("set alarm for 7 p.m."), Cmd.SetAlarm(19, 0))
    check("alarm six thirty", CommandParser.parse("set alarm for six thirty am"), Cmd.SetAlarm(6, 30))
    check("alarm no ampm at 10:00 now", CommandParser.parse("set an alarm for 6", 10, 0), Cmd.SetAlarm(18, 0))
    check("alarm no ampm at 04:00 now", CommandParser.parse("set an alarm for 6", 4, 0), Cmd.SetAlarm(6, 0))
    check("alarm half past", CommandParser.parse("set alarm for half past 7 pm"), Cmd.SetAlarm(19, 30))
    check("alarm quarter to", CommandParser.parse("wake me up at quarter to 7 am"), Cmd.SetAlarm(6, 45))
    check("alarm missing time", CommandParser.parse("set an alarm"), Cmd.Clarify("What time should I set the alarm for, Master?"))
    check("timer 10 min", CommandParser.parse("set a timer for 10 minutes"), Cmd.SetTimer(600))
    check("timer ten minutes", CommandParser.parse("timer for ten minutes"), Cmd.SetTimer(600))
    check("timer 1h30", CommandParser.parse("set timer for 1 hour 30 minutes"), Cmd.SetTimer(5400))
    check("timer half hour", CommandParser.parse("set a timer for half an hour"), Cmd.SetTimer(1800))
    check("timer 45 sec", CommandParser.parse("start a timer for 45 seconds"), Cmd.SetTimer(45))
    check("time", CommandParser.parse("What's the time?"), Cmd.TellTime)
    check("time2", CommandParser.parse("what time is it"), Cmd.TellTime)
    check("date", CommandParser.parse("what's the date today"), Cmd.TellDate)
    check("day", CommandParser.parse("What day is it?"), Cmd.TellDate)
    check("calc plus", CommandParser.parse("what is 12 times 5 plus 3"), Cmd.CalcAnswer("12 times 5 plus 3", "63"))
    check("calc percent", CommandParser.parse("calculate 20 percent of 200"), Cmd.CalcAnswer("20 percent of 200", "40"))
    check("calc div", CommandParser.parse("what's 10 divided by 4"), Cmd.CalcAnswer("10 divided by 4", "2.5"))
    check("calc sqrt", CommandParser.parse("what is square root of 144"), Cmd.CalcAnswer("square root of 144", "12"))
    check("calc power", CommandParser.parse("what is 2 to the power of 10"), Cmd.CalcAnswer("2 to the power of 10", "1024"))
    check("calc bare", CommandParser.parse("15 x 15"), Cmd.CalcAnswer("15 x 15", "225"))
    check("calc div zero -> AI", CommandParser.parse("what is 5 divided by 0") is Cmd.Ask, true)
    check("not math", CommandParser.parse("what is the capital of France") is Cmd.Ask, true)
    check("open app", CommandParser.parse("open WhatsApp"), Cmd.OpenApp("whatsapp"))
    check("open youtube", CommandParser.parse("open YouTube please"), Cmd.OpenApp("youtube"))
    check("open chrome", CommandParser.parse("open chrome"), Cmd.OpenApp("chrome"))
    check("open bms", CommandParser.parse("open book my show"), Cmd.BookTickets(""))
    check("yt search", CommandParser.parse("search lofi music on youtube"), Cmd.YouTubeSearch("lofi music", false))
    check("play on yt", CommandParser.parse("play arise song on youtube"), Cmd.YouTubeSearch("arise song", true))
    check("chrome site", CommandParser.parse("open google.com"), Cmd.ChromeOpen("https://google.com"))
    check("chrome site spoken", CommandParser.parse("go to amazon dot in"), Cmd.ChromeOpen("https://amazon.in"))
    check("chrome search", CommandParser.parse("search for best pizza near me in chrome"), Cmd.ChromeSearch("best pizza near me"))
    check("google", CommandParser.parse("google solo leveling season 3"), Cmd.ChromeSearch("solo leveling season 3"))
    check("install", CommandParser.parse("install Instagram"), Cmd.InstallApp("instagram"))
    check("install from play store", CommandParser.parse("install telegram from play store"), Cmd.InstallApp("telegram"))
    check("install app word", CommandParser.parse("download the spotify app from chrome"), Cmd.InstallApp("spotify"))
    check("gmail open", CommandParser.parse("open gmail"), Cmd.GmailOpen)
    check("gmail read", CommandParser.parse("do I have any new emails"), Cmd.GmailRead)
    check("gmail read2", CommandParser.parse("check my gmail"), Cmd.GmailRead)
    check("compose", CommandParser.parse("send an email to john at gmail dot com saying I will be late"), Cmd.GmailCompose("john@gmail.com", "", "i will be late"))
    check("compose name only", CommandParser.parse("compose an email to mom about dinner"), Cmd.GmailCompose("", "dinner", ""))
    check("book tickets", CommandParser.parse("book tickets for Avengers on bookmyshow"), Cmd.BookTickets("avengers"))
    check("book movie tickets", CommandParser.parse("book movie tickets for Kalki"), Cmd.BookTickets("kalki"))
    check("tap", CommandParser.parse("tap Book tickets"), Cmd.Tap("book tickets"))
    check("tap the", CommandParser.parse("click on the install button"), Cmd.Tap("install button"))
    check("scroll", CommandParser.parse("scroll down"), Cmd.Scroll(true))
    check("scroll up", CommandParser.parse("scroll up"), Cmd.Scroll(false))
    check("type", CommandParser.parse("type Avengers Endgame"), Cmd.TypeText("avengers endgame"))
    check("enter", CommandParser.parse("press enter"), Cmd.PressEnter)
    check("back", CommandParser.parse("go back"), Cmd.Back)
    check("home", CommandParser.parse("go home"), Cmd.Home)
    check("read screen", CommandParser.parse("read the screen"), Cmd.ReadScreen)
    check("stop", CommandParser.parse("stop"), Cmd.Stop)
    check("sleep", CommandParser.parse("go to sleep"), Cmd.Sleep)
    check("ai question", CommandParser.parse("What is the tallest mountain") is Cmd.Ask, true)
    check("ai write poem", CommandParser.parse("write a poem about the moon") is Cmd.Ask, true)

    // wake word
    check("wake hey", WakeMatcher.match("hey jin woo", true)?.remaining, "")
    check("wake bare", WakeMatcher.match("Jin Woo open youtube", false)?.remaining, "open youtube")
    check("wake bare required hey -> no", WakeMatcher.match("jin woo open youtube", true), null)
    check("wake mishear", WakeMatcher.match("hey gin woo what time is it", true)?.remaining, "what time is it")
    check("wake merged", WakeMatcher.match("jinwoo", false)?.remaining, "")
    check("wake sung", WakeMatcher.match("sung jin woo", false)?.remaining, "")
    check("wake none", WakeMatcher.match("what is the weather", false), null)
    check("wake none2", WakeMatcher.match("I need to gym wood", false), null)
    check("yes", WakeMatcher.yesNo("yes please do it"), true)
    check("no", WakeMatcher.yesNo("no don't"), false)
    check("unclear", WakeMatcher.yesNo("banana"), null)

    // guard
    check("block pay", Guard.isBlockedTarget("Pay ₹450"), true)
    check("block proceed to pay", Guard.isBlockedTarget("Proceed to Pay"), true)
    check("block price button", Guard.isBlockedTarget("₹ 149"), true)
    check("block buy", Guard.isBlockedTarget("Buy now"), true)
    check("block uninstall", Guard.isBlockedTarget("Uninstall"), true)
    check("allow install", Guard.isBlockedTarget("Install"), false)
    check("allow book tickets", Guard.isBlockedTarget("Book tickets"), false)
    check("allow select seats", Guard.isBlockedTarget("Select seats"), false)
    check("allow display", Guard.isBlockedTarget("Display settings"), false)
    check("sensitive field", Guard.isSensitiveField("Enter OTP"), true)
    check("normal field", Guard.isSensitiveField("Search"), false)

    check("play yt", CommandParser.parse("play believer"), Cmd.YouTubeSearch("believer", true))
    check("open yt and play", CommandParser.parse("open youtube and play arijit singh songs"), Cmd.YouTubeSearch("arijit singh songs", true))
    check("story", CommandParser.parse("tell me a story about a dragon") is Cmd.Story, true)
    check("play story", CommandParser.parse("play a story") is Cmd.Story, true)
    check("bedtime story", CommandParser.parse("tell me a short bedtime story") is Cmd.Story, true)
    check("call", CommandParser.parse("call mom"), Cmd.Call("mom"))
    check("call make", CommandParser.parse("make a call to Rahul Sharma"), Cmd.Call("rahul sharma"))
    check("wa full", CommandParser.parse("send a whatsapp message to Rahul saying I will be late"), Cmd.WhatsApp("rahul", "i will be late"))
    check("wa short", CommandParser.parse("message Priya on whatsapp saying happy birthday"), Cmd.WhatsApp("priya", "happy birthday"))
    check("wa nobody", CommandParser.parse("send message to dad"), Cmd.WhatsApp("dad", ""))
    check("wa open", CommandParser.parse("whatsapp mom"), Cmd.WhatsApp("mom", ""))
    check("wake arise", WakeMatcher.match("arise", false)?.remaining, "")
    check("wake hey arise", WakeMatcher.match("hey arise open youtube", true)?.remaining, "open youtube")
    check("wake a rise", WakeMatcher.match("a rise what time is it", false)?.remaining, "what time is it")
    check("wake arise needs hey", WakeMatcher.match("arise", true), null)
    check("allow whatsapp send", Guard.isBlockedTarget("Send"), false)
    println(if (fails == 0) "\nALL TESTS PASSED" else "\n$fails FAILED")
}
